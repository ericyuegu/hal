    prefetch.stage_next();update+=1
    return result


def cpu_tree(value):
    if isinstance(value,torch.Tensor):return value.detach().cpu().clone()
    if isinstance(value,dict):return {key:cpu_tree(item) for key,item in value.items()}
    if isinstance(value,(list,tuple)):return type(value)(cpu_tree(item) for item in value)
    return value

def compare_tree(actual,expected,path,stats):
    if isinstance(expected,torch.Tensor):
        assert actual.shape==expected.shape and actual.dtype==expected.dtype,path
        same=torch.equal(actual,expected)
        stats['exact_tensors']+=int(same);stats['tensors']+=1
        if actual.is_floating_point():
            diff=float((actual-expected).abs().max()) if actual.numel() else 0.
            group='gradient' if '/gradients/' in path else 'parameter' if '/model/' in path else 'other'
            stats[group+'_max_abs']=max(stats.get(group+'_max_abs',0.),diff)
            atol,rtol=(.0002,.01) if group=='gradient' else (.000002,.00002)
            torch.testing.assert_close(actual,expected,atol=atol,rtol=rtol,msg=path)
        else:assert same,path
    elif isinstance(expected,dict):
        assert actual.keys()==expected.keys(),path
        for key in expected:compare_tree(actual[key],expected[key],path+'/'+str(key),stats)
    elif isinstance(expected,(tuple,list)):
        assert type(actual) is type(expected) and len(actual)==len(expected),path
        for index,(a,b) in enumerate(zip(actual,expected,strict=True)):compare_tree(a,b,path+'/'+str(index),stats)
    else:assert actual==expected,(path,actual,expected)
