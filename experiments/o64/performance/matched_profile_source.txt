image=(modal.Image.from_registry('ghcr.io/ericyuegu/hal:cuda13')
       .add_local_dir(SOURCE,'/opt/hal',copy=True,ignore=modal.FilePatternMatcher.from_file(SOURCE/'.dockerignore'))

@app.function(image=image,gpu='B200',cpu=(32,48),memory=(131072,393216),ephemeral_disk=524288,timeout=3600,retries=0,serialized=True,include_source=False)
def benchmark(case_source,sha):
    import os
    env=os.environ.copy()
    for name,suffix in [('TORCHINDUCTOR_CACHE_DIR','torchinductor'),('TRITON_CACHE_DIR','triton'),('CUDA_CACHE_PATH','cuda'),('TMPDIR','tmp')]:
        path=Path('/opt/hal-local-cache')/suffix;path.mkdir(parents=True,exist_ok=True);env[name]=str(path)
    result={'schema_version':1,'git_sha':sha,'dataset_downloads':False,'cases':[],'failures':[]}
    for label,name,mode,profile in [('059','059','reduce-overhead','no'),('063','063','reduce-overhead','no'),
                                  ('064','064','reduce-overhead','yes'),('064-max-autotune','064','max-autotune','no'),
                                  ('059-repeat','059','reduce-overhead','no')]:
        print('MATCHED_START',label,flush=True)
        child=subprocess.run(['/opt/venv/bin/python','-c',case_source,name,mode,profile],cwd='/opt/hal',env=env,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
        print(child.stdout,flush=True)
        if child.returncode:
            result['failures'].append({'label':label,'exit_code':child.returncode,'output':child.stdout})
            break
        lines=[line.removeprefix('MATCHED_RESULT ') for line in child.stdout.splitlines() if line.startswith('MATCHED_RESULT ')]
        if len(lines)!=1:
            raise RuntimeError('missing benchmark result for '+label)
        case=json.loads(lines[0]);case['label']=label;result['cases'].append(case)
    result['status']='passed' if not result['failures'] and len(result['cases'])==5 else 'failed'
    return result

with app.run(detach=True):

    OUTPUT.parent.mkdir(parents=True,exist_ok=True)
    OUTPUT.write_text(json.dumps({'app_id':app.app_id,'function_call_id':call.object_id,'git_sha':SHA,
                                 'benchmark_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()},indent=2)+'\n')
    print('SUBMITTED',app.app_id,call.object_id,flush=True)
